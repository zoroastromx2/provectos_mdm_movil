```cpp
struct CreateGpkgResult {
    bool success;
    QString errorMsg;
    QString gpkgPath;
    QStringList layerNames;
};

// geomanager.cpp (método público)
bool GeoManager::createGeoPackage(const QStringList &shpPaths, 
                                   const QString &gpkgPath)
{
    setBusy(true);

    // ? LANZA EN HILO DE TRABAJO
    QtConcurrent::run([this, shpPaths, gpkgPath]() {
        return createGeoPackageWorker(shpPaths, gpkgPath);
    });

    return true; // INMEDIATO
}

// Worker (ejecuta en hilo de trabajo)
CreateGpkgResult GeoManager::createGeoPackageWorker(...)
{
    CreateGpkgResult result;

    // Abre SHP
    // Copia capas
    // Escribe GPKG
    // ... (15 segundos sin afectar UI)

    return result;
}

// Handler (ejecuta en UI thread cuando termina)
void GeoManager::onCreateGeoPackageFinished()
{
    const CreateGpkgResult result = m_createGpkgWatcher->result();

    if (result.success) {
        setActiveGpkgPath(result.gpkgPath);
        m_layerNames = result.layerNames;
        emit layerNamesChanged();
        emit operationSucceeded();
    } else {
        setLastError(result.errorMsg);
    }

    setBusy(false);
}
```

## ??? Técnicas Utilizadas

### QtConcurrent
```cpp
// Antes:
bool result = expensiveOperation();  // Bloquea

// Después:
QFuture<ResultType> future = QtConcurrent::run(
    [capture] { return expensiveOperation(); }
);

QFutureWatcher<ResultType> *watcher = new QFutureWatcher();
connect(watcher, &QFutureWatcher<ResultType>::finished, 
        this, &Class::onOperationFinished);
watcher->setFuture(future);
```

### Thread-Safety
```cpp
// ? Seguro
struct Result { bool success; QString data; };  // Copiable
QtConcurrent::run([paths, gpkg] {  // Capturado por valor
    return expensiveOp(paths, gpkg);  // Copias, sin compartir
});

// ?? Inseguro (no hecho)
QtConcurrent::run([&paths] {  // Captura por referencia
    return expensiveOp(paths);  // Podría modificarse en UI thread
});
```
